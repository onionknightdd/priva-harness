heads=[]
for i,l in enumerate(lines):
 m=re.match(r'^(#{1,6}) (.+)$',l)
 if m: heads.append((i,len(m[1]),m[2]))
def block(title, occurrence=0):
 found=[h for h in heads if h[2]==title][occurrence]
 stop=next((h[0] for h in heads if h[0]>found[0] and h[1]<=found[1]),len(lines))
 return '\n'.join(lines[found[0]+1:stop]).strip()
def subblock(title,start_title,end_title=None):
 s=block(title)
 start=s.index(start_title)
 return s[start:s.index(end_title,start) if end_title else None]
def clean(s):return s.replace('`','').strip()
def field(path,typ,desc,required=False,default=None,children=None):
 name=re.sub(r'^.*\.(?![^\[]*\])','',path)
 o=dict(path=path,name=name,type=typ,required=required,description=desc)
 if default is not None:o['default']=str(default)
 if children:o['children']=children
 return o
def table_fields(text):
 out=[]
 for l in text.splitlines():
  if not l.startswith('|'):continue
  cells=[c.strip() for c in l.strip('|').split('|')]
  if len(cells)<3 or re.fullmatch(r'[- :]+',cells[0]) or not cells[0].startswith('`'):continue
  path=clean(cells[0]); typ=clean(cells[1])
  if not re.match(r'^(request|response|frame|state)\.',path): continue
  cond=clean(cells[2]) if len(cells)==4 else ''
  desc=cells[3] if len(cells)==4 else cells[2]
  required=cond in ['是','必填','必有','本产品必有','本产品成功执行请求必有']
  if cond and not required:desc+=' 出现条件：'+cond+'。'
  o=field(path,typ,desc,required)
  m=re.search(r'默认\s*`?([^`，；。 ]+)',cond)
  if m:o['default']=m[1]
  out.append(o)
 return out
def nested(fs):
 out=[]
 for f in fs:
  f=copy.deepcopy(f)
  parents=[x for x in fs if x is not f and f['path']!=x['path'] and (f['path'].startswith(x['path']+'.') or f['path'].startswith(x['path']+'['))]
  parent=max(parents,key=lambda x:len(x['path'])) if parents else None
  f['_parent']=parent['path'] if parent else None
  out.append(f)
 lookup={f['path']:f for f in out}; roots=[]
 for f in out:
  p=f.pop('_parent')
  if p in lookup: lookup[p].setdefault('children',[]).append(f)
  else:roots.append(f)
 return roots
def prose(text):
 # Field tables are represented by the structured field list; preserve prose and code diagrams.
 result=[]; in_table=False; field_table=False
 for l in text.splitlines():
  if l.startswith('|'):
   if not in_table:
    field_table=any(k in l for k in ['完整字段路径','返回路径','请求头','位置与名称','参数 |','Query 参数'])
   in_table=True
   if not field_table:result.append(l)
  else:
   in_table=False
   result.append(l)
 return '\n'.join(result).strip()
def sec(id,title,text,fields=None):
 fs=table_fields(text) if fields is None else fields
 body=prose(text)
 o={'id':id,'title':title}
 if fs:o['fields']=nested(fs)
 if body:o['body']=body
 return o
